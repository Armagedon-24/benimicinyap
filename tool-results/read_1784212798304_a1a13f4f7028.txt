    75→  const { currentUser, tasks, setTasks, addLog, systemLogs, addLog: storeAddLog } = useAppStore();
    76→
    77→  const [users, setUsers] = useState<User[]>([]);
    78→  const [allTasks, setAllTasks] = useState<Task[]>([]);
    79→  const [stats, setStats] = useState<any>(null);
    80→  const [loading, setLoading] = useState(true);
    81→  const [seedLoading, setSeedLoading] = useState(false);
    82→  const logEndRef = useRef<HTMLDivElement>(null);
    83→
    84→  // Revenue system state
    85→  const [revenue, setRevenue] = useState<any>(null);
    86→  const [settings, setSettings] = useState<any[]>([]);
    87→  const [revenueLoading, setRevenueLoading] = useState(true);
    88→  const [settingsSaving, setSettingsSaving] = useState(false);
    89→
    90→  // Local settings form
    91→  const [commissionRate, setCommissionRate] = useState(10);
    92→  const [commissionMode, setCommissionMode] = useState('percentage');
    93→  const [minCommission, setMinCommission] = useState(10);
    94→  const [maxCommission, setMaxCommission] = useState(500);
    95→  const [commissionPayer, setCommissionPayer] = useState('tasker');
    96→  const [listingFee, setListingFee] = useState(0);
    97→  const [listingFeeEnabled, setListingFeeEnabled] = useState(false);
    98→  const [withdrawFeeRate, setWithdrawFeeRate] = useState(1);
    99→  const [minWithdrawFee, setMinWithdrawFee] = useState(5);
   100→  const [maxWithdrawFee, setMaxWithdrawFee] = useState(100);
   101→  const [withdrawFeeEnabled, setWithdrawFeeEnabled] = useState(true);
   102→
   103→  const formatTL = (amount: number) =>
   104→    new Intl.NumberFormat('tr-TR').format(amount) + ' ₺';
   105→
   106→  const formatDate = (date: string) =>
   107→    date ? new Date(date).toLocaleDateString('tr-TR') : '';
   108→
   109→  const fetchAll = useCallback(async () => {
   110→    setLoading(true);
   111→    storeAddLog('Admin verileri yenileniyor...');
   112→    try {
   113→      const [usersRes, tasksRes, statsRes, settingsRes, revenueRes] = await Promise.all([
   114→        fetch('/api/admin?action=users'),
   115→        fetch('/api/tasks'),
   116→        fetch('/api/admin?action=stats'),
   117→        fetch('/api/admin/settings'),
   118→        fetch('/api/admin/revenue'),
   119→      ]);
   120→      const [usersData, tasksData, statsData, settingsData, revenueData] = await Promise.all([
   121→        usersRes.json(),
   122→        tasksRes.json(),
   123→        statsRes.json(),
   124→        settingsRes.json(),
   125→        revenueRes.json(),
   126→      ]);
   127→      if (usersData.success) setUsers(usersData.users);
   128→      if (tasksData.success) {
   129→        setAllTasks(tasksData.tasks);
   130→        setTasks(tasksData.tasks);
   131→      }
   132→      if (statsData.success) setStats(statsData.stats);
   133→      if (settingsData.success) {
   134→        setSettings(settingsData.settings);
   135→        applySettingsToLocal(settingsData.settings);
   136→      }
   137→      if (revenueData.success) setRevenue(revenueData.revenue);
   138→      storeAddLog('Admin verileri başarıyla yüklendi');
   139→    } catch {
   140→      storeAddLog('Admin veri yükleme hatası');
   141→    } finally {
   142→      setLoading(false);
   143→      setRevenueLoading(false);
   144→    }
   145→  }, [setTasks, storeAddLog]);
   146→
   147→  const applySettingsToLocal = (s: any[]) => {
   148→    const m: Record<string, string> = {};
   149→    s.forEach((item) => { m[item.key] = item.value; });
   150→    if (m['commission_rate'] !== undefined) setCommissionRate(parseFloat(m['commission_rate']));
   151→    if (m['commission_mode'] !== undefined) setCommissionMode(m['commission_mode']);
   152→    if (m['min_commission'] !== undefined) setMinCommission(parseFloat(m['min_commission']));
   153→    if (m['max_commission'] !== undefined) setMaxCommission(parseFloat(m['max_commission']));
   154→    if (m['commission_payer'] !== undefined) setCommissionPayer(m['commission_payer']);
   155→    if (m['listing_fee'] !== undefined) setListingFee(parseFloat(m['listing_fee']));
   156→    if (m['listing_fee_enabled'] !== undefined) setListingFeeEnabled(m['listing_fee_enabled'] === 'true');
   157→    if (m['withdraw_fee_rate'] !== undefined) setWithdrawFeeRate(parseFloat(m['withdraw_fee_rate']));
   158→    if (m['min_withdraw_fee'] !== undefined) setMinWithdrawFee(parseFloat(m['min_withdraw_fee']));
   159→    if (m['max_withdraw_fee'] !== undefined) setMaxWithdrawFee(parseFloat(m['max_withdraw_fee']));
   160→    if (m['withdraw_fee_enabled'] !== undefined) setWithdrawFeeEnabled(m['withdraw_fee_enabled'] === 'true');
   161→  };
   162→
   163→  useEffect(() => {
   164→    fetchAll();
   165→  }, [fetchAll]);
   166→
   167→  useEffect(() => {
   168→    logEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   169→  }, [systemLogs]);
   170→
   171→  const handleSeed = async () => {
   172→    setSeedLoading(true);
   173→    storeAddLog('DB sıfırlama başlatıldı...');
   174→    try {
   175→      const res = await fetch('/api/seed', { method: 'POST' });
   176→      const data = await res.json();
   177→      if (data.success) {
   178→        toast.success(data.message);
   179→        storeAddLog('DB sıfırlandı ve yeniden dolduruldu');
   180→        await fetchAll();
   181→      } else {
   182→        toast.error(data.message || 'Hata oluştu');
   183→      }
   184→    } catch {
   185→      toast.error('Bağlantı hatası');
   186→    } finally {
   187→      setSeedLoading(false);
   188→    }
   189→  };
   190→
   191→  const handleSaveSettings = async () => {
   192→    setSettingsSaving(true);
   193→    try {
   194→      const payload = [
   195→        { key: 'commission_rate', value: String(commissionRate) },
   196→        { key: 'commission_mode', value: commissionMode },
   197→        { key: 'min_commission', value: String(minCommission) },
   198→        { key: 'max_commission', value: String(maxCommission) },
   199→        { key: 'commission_payer', value: commissionPayer },
   200→        { key: 'listing_fee', value: String(listingFee) },
   201→        { key: 'listing_fee_enabled', value: String(listingFeeEnabled) },
   202→        { key: 'withdraw_fee_rate', value: String(withdrawFeeRate) },
   203→        { key: 'min_withdraw_fee', value: String(minWithdrawFee) },
   204→        { key: 'max_withdraw_fee', value: String(maxWithdrawFee) },
   205→        { key: 'withdraw_fee_enabled', value: String(withdrawFeeEnabled) },
   206→      ];
   207→      const res = await fetch('/api/admin/settings', {
   208→        method: 'PUT',
   209→        headers: { 'Content-Type': 'application/json' },
   210→        body: JSON.stringify({ settings: payload }),
   211→      });
   212→      const data = await res.json();
   213→      if (data.success) {
   214→        toast.success('Komisyon ayarları kaydedildi');
   215→        storeAddLog('Komisyon ayarları güncellendi');
   216→      } else {
   217→        toast.error(data.message);
   218→      }
   219→    } catch {
   220→      toast.error('Ayarlar kaydedilirken hata oluştu');
   221→    } finally {
   222→      setSettingsSaving(false);
   223→    }
   224→  };
   225→
   226→  const handleToggleTC = async (user: User) => {
   227→    try {
   228→      const res = await fetch('/api/admin', {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({ action: 'adjust-balance', userId: user.id, amount: 0 }),
   232→      });
   233→      // TC toggle is simulated via the adjust balance - we'll toggle locally
   234→      setUsers((prev) =>
   235→        prev.map((u) => (u.id === user.id ? { ...u, tcVerified: !u.tcVerified } : u))
   236→      );
   237→      storeAddLog(`TC doğrulama değiştirildi: ${user.fullName} → ${!user.tcVerified}`);
   238→      toast.success(`${user.fullName} T.C. doğrulama güncellendi`);
   239→    } catch {
   240→      toast.error('Hata oluştu');
   241→    }
   242→  };
   243→
   244→  const handleAdjustBalance = async (userId: string, amount: number) => {
   245→    if (isNaN(amount) || amount === 0) {
   246→      toast.error('Geçerli bir tutar girin');
   247→      return;
   248→    }
   249→    try {
   250→      const res = await fetch('/api/admin', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ action: 'adjust-balance', userId, amount }),
   254→      });
   255→      const data = await res.json();
   256→      if (data.success) {
   257→        setUsers((prev) =>
   258→          prev.map((u) => (u.id === userId ? { ...u, walletBalance: data.user.walletBalance } : u))
   259→        );
   260→        storeAddLog(`Bakiye ayarlandı: ${userId} → ${amount > 0 ? '+' : ''}${formatTL(amount)}`);
   261→        toast.success(`Bakiye güncellendi: ${amount > 0 ? '+' : ''}${formatTL(amount)}`);
   262→      } else {
   263→        toast.error(data.message);
   264→      }
   265→    } catch {
   266→      toast.error('Hata oluştu');
   267→    }
   268→  };
   269→
   270→  const handleDeleteUser = async (userId: string, fullName: string) => {
   271→    try {
   272→      const res = await fetch(`/api/admin?id=${userId}`, { method: 'DELETE' });
   273→      const data = await res.json();
   274→      if (data.success) {
   275→        setUsers((prev) => prev.filter((u) => u.id !== userId));
   276→        storeAddLog(`Kullanıcı silindi: ${fullName}`);
   277→        toast.success(`${fullName} silindi`);
   278→        await fetchAll();
   279→      } else {
   280→        toast.error(data.message);
   281→      }
   282→    } catch {
   283→      toast.error('Hata oluştu');
   284→    }
   285→  };
   286→
   287→  const handleTaskStatus = async (taskId: string, status: string, assignedToId?: string, assignedToName?: string) => {
   288→    try {
   289→      const body: any = { action: 'task-status', taskId, status };
   290→      if (assignedToId) body.assignedToId = assignedToId;
   291→      if (assignedToName) body.assignedToName = assignedToName;
   292→
   293→      const res = await fetch('/api/admin', {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify(body),
   297→      });
   298→      const data = await res.json();
   299→      if (data.success) {
   300→        storeAddLog(`Görev durumu değiştirildi: ${taskId} → ${status}`);
   301→        toast.success('Görev durumu güncellendi');
   302→        await fetchAll();
   303→      } else {
   304→        toast.error(data.message);
   305→      }
   306→    } catch {
   307→      toast.error('Hata oluştu');
   308→    }
   309→  };
   310→
   311→  if (loading) {
   312→    return (
   313→      <div className="flex items-center justify-center py-20">
   314→        <Loader2 className="w-8 h-8 animate-spin text-blue-500" />
   315→      </div>
   316→    );
   317→  }
   318→
   319→  return (
   320→    <div className="space-y-5">
   321→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   322→        <div className="flex items-center justify-between flex-wrap gap-3">
   323→          <div>
   324→            <h2 className="text-xl font-bold flex items-center gap-2">
   325→              <ShieldCheck className="w-6 h-6 text-blue-600" />
   326→              Admin Paneli
   327→            </h2>
   328→            <p className="text-sm text-muted-foreground mt-1">
   329→              Platform yönetimi ve izleme
   330→            </p>
   331→          </div>
   332→          <div className="flex gap-2">
   333→            <Button variant="outline" size="sm" onClick={fetchAll}>
   334→              <RefreshCw className="w-3.5 h-3.5 mr-1.5" />
   335→              Verileri Yenile
   336→            </Button>
   337→            <AlertDialog>
   338→              <AlertDialogTrigger asChild>
   339→                <Button variant="destructive" size="sm" disabled={seedLoading}>
   340→                  {seedLoading ? (
   341→                    <Loader2 className="w-3.5 h-3.5 mr-1.5 animate-spin" />
   342→                  ) : (
   343→                    <Database className="w-3.5 h-3.5 mr-1.5" />
   344→                  )}
   345→                  DB Sıfırla
   346→                </Button>
   347→              </AlertDialogTrigger>
   348→              <AlertDialogContent>
   349→                <AlertDialogHeader>
   350→                  <AlertDialogTitle>Veritabanı Sıfırla</AlertDialogTitle>
   351→                  <AlertDialogDescription>
   352→                    Tüm veriler silinecek ve demo veriler yeniden oluşturulacak. Bu işlem geri
   353→                    alınamaz. Emin misiniz?
   354→                  </AlertDialogDescription>
   355→                </AlertDialogHeader>
   356→                <AlertDialogFooter>
   357→                  <AlertDialogCancel>İptal</AlertDialogCancel>
   358→                  <AlertDialogAction onClick={handleSeed}>Sıfırla</AlertDialogAction>
   359→                </AlertDialogFooter>
   360→              </AlertDialogContent>
   361→            </AlertDialog>
   362→          </div>
   363→        </div>
   364→      </motion.div>
   365→
   366→      <Tabs defaultValue="overview" className="w-full">
   367→        <TabsList className="grid w-full grid-cols-6 max-w-4xl overflow-x-auto">
   368→          <TabsTrigger value="overview" className="text-xs">
   369→            Genel Durum
   370→          </TabsTrigger>
   371→          <TabsTrigger value="users" className="text-xs">
   372→            Kullanıcılar
   373→          </TabsTrigger>
   374→          <TabsTrigger value="tasks" className="text-xs">
   375→            Görev & Güvence
   376→          </TabsTrigger>
   377→          <TabsTrigger value="revenue" className="text-xs">
   378→            Gelir Sistemi
   379→          </TabsTrigger>
   380→          <TabsTrigger value="console" className="text-xs">
   381→            Sistem Konsolu
   382→          </TabsTrigger>
   383→          <TabsTrigger value="admin-info" className="text-xs">
   384→            Yönetici
   385→          </TabsTrigger>
   386→        </TabsList>
   387→
   388→        {/* Overview Tab */}
   389→        <TabsContent value="overview" className="mt-5 space-y-5">
   390→          {stats && (
   391→            <>
   392→              <div className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   393→                {[
   394→                  { label: 'Kullanıcı', value: stats.totalUsers, icon: Users, color: 'text-blue-600' },
   395→                  { label: 'Görev', value: stats.totalTasks, icon: ClipboardList, color: 'text-purple-600' },
   396→                  { label: 'Bekleyen', value: stats.pendingTasks, icon: Info, color: 'text-amber-600' },
   397→                  { label: 'Tamamlanan', value: stats.completedTasks, icon: UserCheck, color: 'text-green-600' },
   398→                  { label: 'İşlem Hacmi', value: formatTL(stats.totalVolume), icon: DollarSign, color: 'text-red-600' },
   399→                ].map((item) => (
   400→                  <Card key={item.label}>
   401→                    <CardContent className="p-4 text-center">
   402→                      <item.icon className={`w-5 h-5 ${item.color} mx-auto mb-1.5`} />
   403→                      <p className="text-xs text-muted-foreground">{item.label}</p>
   404→                      <p className="text-lg font-bold mt-0.5">{item.value}</p>
   405→                    </CardContent>
   406→                  </Card>
   407→                ))}
   408→              </div>
   409→
   410→              {/* Financial summary */}
   411→              <Card>
   412→                <CardHeader className="pb-2">
   413→                  <CardTitle className="text-sm">Finansal Özet</CardTitle>
   414→                </CardHeader>
   415→                <CardContent>
   416→                  <div className="grid grid-cols-2 gap-4">
   417→                    <div className="p-3 rounded-lg bg-green-50 dark:bg-green-950/30">
   418→                      <p className="text-xs text-muted-foreground">Toplam İşlem Hacmi</p>
   419→                      <p className="text-lg font-bold text-green-600 dark:text-green-400">
   420→                        {formatTL(stats.totalVolume)}
   421→                      </p>
   422→                    </div>
   423→                    <div className="p-3 rounded-lg bg-amber-50 dark:bg-amber-950/30">
   424→                      <p className="text-xs text-muted-foreground">Escrow Hacmi</p>
   425→                      <p className="text-lg font-bold text-amber-600 dark:text-amber-400">
   426→                        {formatTL(stats.escrowVolume)}
   427→                      </p>
   428→                    </div>
   429→                  </div>
   430→                </CardContent>
   431→              </Card>
   432→            </>
   433→          )}
   434→        </TabsContent>
   435→
   436→        {/* Users Tab */}
   437→        <TabsContent value="users" className="mt-5">
   438→          <Card>
   439→            <CardContent className="p-0">
   440→              <div className="max-h-[500px] overflow-y-auto">
   441→                <Table>
   442→                  <TableHeader>
   443→                    <TableRow>
   444→                      <TableHead className="text-xs">Kullanıcı</TableHead>
   445→                      <TableHead className="text-xs">Rol</TableHead>
   446→                      <TableHead className="text-xs">T.C.</TableHead>
   447→                      <TableHead className="text-xs">Bakiye</TableHead>
   448→                      <TableHead className="text-xs">Şehir</TableHead>
   449→                      <TableHead className="text-xs text-right">İşlemler</TableHead>
   450→                    </TableRow>
   451→                  </TableHeader>
   452→                  <TableBody>
   453→                    {users.map((user) => (
   454→                      <TableRow key={user.id}>
   455→                        <TableCell>
   456→                          <div>
   457→                            <p className="text-sm font-medium">{user.fullName}</p>
   458→                            <p className="text-xs text-muted-foreground">{user.email}</p>
   459→                          </div>
   460→                        </TableCell>
   461→                        <TableCell>
   462→                          <Badge variant={user.role === 'admin' ? 'destructive' : 'secondary'} className="text-[10px]">
   463→                            {user.role === 'admin' ? 'Admin' : 'Üye'}
   464→                          </Badge>
   465→                        </TableCell>
   466→                        <TableCell>
   467→                          <Switch
   468→                            checked={user.tcVerified}
   469→                            onCheckedChange={() => handleToggleTC(user)}
   470→                            className="data-[state=checked]:bg-green-500"
   471→                          />
   472→                        </TableCell>
   473→                        <TableCell className="text-sm font-medium">
   474→                          {formatTL(user.walletBalance)}
   475→                        </TableCell>
   476→                        <TableCell className="text-sm text-muted-foreground">{user.city}</TableCell>
   477→                        <TableCell>
   478→                          <div className="flex items-center justify-end gap-1">
   479→                            <BalanceAdjustButton userId={user.id} onAdjust={handleAdjustBalance} />
   480→                            {user.role !== 'admin' && (
   481→                              <AlertDialog>
   482→                                <AlertDialogTrigger asChild>
   483→                                  <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500">
   484→                                    <Trash2 className="w-3.5 h-3.5" />
   485→                                  </Button>
   486→                                </AlertDialogTrigger>
   487→                                <AlertDialogContent>
   488→                                  <AlertDialogHeader>
   489→                                    <AlertDialogTitle>Kullanıcı Sil</AlertDialogTitle>
   490→                                    <AlertDialogDescription>
   491→                                      <strong>{user.fullName}</strong> kullanıcısını ve tüm
   492→                                      ilişkili verileri silmek istediğinize emin misiniz?
   493→                                    </AlertDialogDescription>
   494→                                  </AlertDialogHeader>
   495→                                  <AlertDialogFooter>
   496→                                    <AlertDialogCancel>İptal</AlertDialogCancel>
   497→                                    <AlertDialogAction onClick={() => handleDeleteUser(user.id, user.fullName)}>
   498→                                      Sil
   499→                                    </AlertDialogAction>
   500→                                  </AlertDialogFooter>
   501→                                </AlertDialogContent>
   502→                              </AlertDialog>
   503→                            )}
   504→                          </div>
   505→                        </TableCell>
   506→                      </TableRow>
   507→                    ))}
   508→                  </TableBody>
   509→                </Table>
   510→              </div>
   511→            </CardContent>
   512→          </Card>
   513→        </TabsContent>
   514→
   515→        {/* Tasks Tab */}
   516→        <TabsContent value="tasks" className="mt-5">
