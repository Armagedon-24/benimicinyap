     2→

     4→import { motion } from 'framer-motion';
     5→import {

    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Badge } from '@/components/ui/badge';
    23→import { Switch } from '@/components/ui/switch';
    24→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    25→import {

    27→  SelectContent,

    30→  SelectValue,
    31→} from '@/components/ui/select';
    32→import {

    34→  TableBody,
    35→  TableCell,
    36→  TableHead,
    37→  TableHeader,
    38→  TableRow,
    39→} from '@/components/ui/table';
    40→import {
    41→  AlertDialog,
    42→  AlertDialogAction,
    43→  AlertDialogCancel,
    44→  AlertDialogContent,
    45→  AlertDialogDescription,
    46→  AlertDialogFooter,
    47→  AlertDialogHeader,
    48→  AlertDialogTitle,
    49→  AlertDialogTrigger,
    50→} from '@/components/ui/alert-dialog';
    51→import { ScrollArea } from '@/components/ui/scroll-area';
    52→import { Separator } from '@/components/ui/separator';
    53→import { toast } from 'sonner';
    54→import { useAppStore } from '@/store/useAppStore';
    55→import type { User, Task } from '@/lib/types';
    56→
    57→const statusMap: Record<string, { label: string; variant: 'default' | 'secondary' | 'destructive' | 'outline' }> = {
    58→  pending: { label: 'Bekliyor', variant: 'secondary' },
    59→  assigned: { label: 'Atandı', variant: 'default' },
    60→  in_progress: { label: 'Devam Ediyor', variant: 'default' },
    61→  completed: { label: 'Tamamlandı', variant: 'outline' },
    62→  cancelled: { label: 'İptal', variant: 'destructive' },
    63→};
    64→
    65→export default function AdminPanel() {
    66→  const { currentUser, tasks, setTasks, addLog, systemLogs, addLog: storeAddLog } = useAppStore();
    67→
    68→  const [users, setUsers] = useState<User[]>([]);
    69→  const [allTasks, setAllTasks] = useState<Task[]>([]);
    70→  const [stats, setStats] = useState<any>(null);
    71→  const [loading, setLoading] = useState(true);
    72→  const [seedLoading, setSeedLoading] = useState(false);
    73→  const logEndRef = useRef<HTMLDivElement>(null);
    74→
    75→  const formatTL = (amount: number) =>
    76→    new Intl.NumberFormat('tr-TR').format(amount) + ' ₺';
    77→
    78→  const formatDate = (date: string) =>
    79→    date ? new Date(date).toLocaleDateString('tr-TR') : '';
    80→
    81→  const fetchAll = useCallback(async () => {
    82→    setLoading(true);
    83→    storeAddLog('Admin verileri yenileniyor...');
    84→    try {
    85→      const [usersRes, tasksRes, statsRes] = await Promise.all([
    86→        fetch('/api/admin?action=users'),
    87→        fetch('/api/tasks'),
    88→        fetch('/api/admin?action=stats'),
    89→      ]);
    90→      const [usersData, tasksData, statsData] = await Promise.all([
    91→        usersRes.json(),
    92→        tasksRes.json(),
    93→        statsRes.json(),
    94→      ]);
    95→      if (usersData.success) setUsers(usersData.users);
    96→      if (tasksData.success) {
    97→        setAllTasks(tasksData.tasks);
    98→        setTasks(tasksData.tasks);
    99→      }
   100→      if (statsData.success) setStats(statsData.stats);
   101→      storeAddLog('Admin verileri başarıyla yüklendi');
   102→    } catch {
   103→      storeAddLog('Admin veri yükleme hatası');
   104→    } finally {
   105→      setLoading(false);
   106→    }
   107→  }, [setTasks, storeAddLog]);
   108→
   109→  useEffect(() => {
   110→    fetchAll();
   111→  }, [fetchAll]);
   112→
   113→  useEffect(() => {
   114→    logEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   115→  }, [systemLogs]);
   116→
   117→  const handleSeed = async () => {
   118→    setSeedLoading(true);
   119→    storeAddLog('DB sıfırlama başlatıldı...');
   120→    try {
   121→      const res = await fetch('/api/seed', { method: 'POST' });
   122→      const data = await res.json();
   123→      if (data.success) {
   124→        toast.success(data.message);
   125→        storeAddLog('DB sıfırlandı ve yeniden dolduruldu');
   126→        await fetchAll();
   127→      } else {
   128→        toast.error(data.message || 'Hata oluştu');
   129→      }
   130→    } catch {
   131→      toast.error('Bağlantı hatası');
   132→    } finally {
   133→      setSeedLoading(false);
   134→    }
   135→  };
   136→
   137→  const handleToggleTC = async (user: User) => {
   138→    try {
   139→      const res = await fetch('/api/admin', {
   140→        method: 'POST',
   141→        headers: { 'Content-Type': 'application/json' },
   142→        body: JSON.stringify({ action: 'adjust-balance', userId: user.id, amount: 0 }),
   143→      });
   144→      // TC toggle is simulated via the adjust balance - we'll toggle locally
   145→      setUsers((prev) =>
   146→        prev.map((u) => (u.id === user.id ? { ...u, tcVerified: !u.tcVerified } : u))
   147→      );
   148→      storeAddLog(`TC doğrulama değiştirildi: ${user.fullName} → ${!user.tcVerified}`);
   149→      toast.success(`${user.fullName} T.C. doğrulama güncellendi`);
   150→    } catch {
   151→      toast.error('Hata oluştu');
   152→    }
   153→  };
   154→
   155→  const handleAdjustBalance = async (userId: string, amount: number) => {
   156→    if (isNaN(amount) || amount === 0) {
   157→      toast.error('Geçerli bir tutar girin');
   158→      return;
   159→    }
   160→    try {
   161→      const res = await fetch('/api/admin', {
   162→        method: 'POST',
   163→        headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify({ action: 'adjust-balance', userId, amount }),
   165→      });
   166→      const data = await res.json();
   167→      if (data.success) {
   168→        setUsers((prev) =>
   169→          prev.map((u) => (u.id === userId ? { ...u, walletBalance: data.user.walletBalance } : u))
   170→        );
   171→        storeAddLog(`Bakiye ayarlandı: ${userId} → ${amount > 0 ? '+' : ''}${formatTL(amount)}`);
   172→        toast.success(`Bakiye güncellendi: ${amount > 0 ? '+' : ''}${formatTL(amount)}`);
   173→      } else {
   174→        toast.error(data.message);
   175→      }
   176→    } catch {
   177→      toast.error('Hata oluştu');
   178→    }
   179→  };
   180→
   181→  const handleDeleteUser = async (userId: string, fullName: string) => {
   182→    try {
   183→      const res = await fetch(`/api/admin?id=${userId}`, { method: 'DELETE' });
   184→      const data = await res.json();
   185→      if (data.success) {
   186→        setUsers((prev) => prev.filter((u) => u.id !== userId));
   187→        storeAddLog(`Kullanıcı silindi: ${fullName}`);
   188→        toast.success(`${fullName} silindi`);
   189→        await fetchAll();
   190→      } else {
   191→        toast.error(data.message);
   192→      }
   193→    } catch {
   194→      toast.error('Hata oluştu');
   195→    }
   196→  };
   197→
   198→  const handleTaskStatus = async (taskId: string, status: string, assignedToId?: string, assignedToName?: string) => {
   199→    try {
   200→      const body: any = { action: 'task-status', taskId, status };
   201→      if (assignedToId) body.assignedToId = assignedToId;
   202→      if (assignedToName) body.assignedToName = assignedToName;
   203→
   204→      const res = await fetch('/api/admin', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify(body),
   208→      });
   209→      const data = await res.json();
   210→      if (data.success) {
   211→        storeAddLog(`Görev durumu değiştirildi: ${taskId} → ${status}`);
   212→        toast.success('Görev durumu güncellendi');
   213→        await fetchAll();
   214→      } else {
   215→        toast.error(data.message);
   216→      }
   217→    } catch {
   218→      toast.error('Hata oluştu');
   219→    }
   220→  };
   221→
   222→  if (loading) {
   223→    return (
   224→      <div className="flex items-center justify-center py-20">
   225→        <Loader2 className="w-8 h-8 animate-spin text-blue-500" />
   226→      </div>
   227→    );
   228→  }
   229→
   230→  return (
   231→    <div className="space-y-5">
   232→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   233→        <div className="flex items-center justify-between flex-wrap gap-3">
   234→          <div>
   235→            <h2 className="text-xl font-bold flex items-center gap-2">
   236→              <ShieldCheck className="w-6 h-6 text-blue-600" />
   237→              Admin Paneli
   238→            </h2>
   239→            <p className="text-sm text-muted-foreground mt-1">
   240→              Platform yönetimi ve izleme
   241→            </p>
   242→          </div>
   243→          <div className="flex gap-2">
   244→            <Button variant="outline" size="sm" onClick={fetchAll}>
   245→              <RefreshCw className="w-3.5 h-3.5 mr-1.5" />
   246→              Verileri Yenile
   247→            </Button>
   248→            <AlertDialog>
   249→              <AlertDialogTrigger asChild>
   250→                <Button variant="destructive" size="sm" disabled={seedLoading}>
   251→                  {seedLoading ? (
   252→                    <Loader2 className="w-3.5 h-3.5 mr-1.5 animate-spin" />
   253→                  ) : (
   254→                    <Database className="w-3.5 h-3.5 mr-1.5" />
   255→                  )}
   256→                  DB Sıfırla
   257→                </Button>
   258→              </AlertDialogTrigger>
   259→              <AlertDialogContent>
   260→                <AlertDialogHeader>
   261→                  <AlertDialogTitle>Veritabanı Sıfırla</AlertDialogTitle>
   262→                  <AlertDialogDescription>
   263→                    Tüm veriler silinecek ve demo veriler yeniden oluşturulacak. Bu işlem geri
   264→                    alınamaz. Emin misiniz?
   265→                  </AlertDialogDescription>
   266→                </AlertDialogHeader>
   267→                <AlertDialogFooter>
   268→                  <AlertDialogCancel>İptal</AlertDialogCancel>
   269→                  <AlertDialogAction onClick={handleSeed}>Sıfırla</AlertDialogAction>
   270→                </AlertDialogFooter>
   271→              </AlertDialogContent>
   272→            </AlertDialog>
   273→          </div>
   274→        </div>
   275→      </motion.div>
   276→
   277→      <Tabs defaultValue="overview" className="w-full">
   278→        <TabsList className="grid w-full grid-cols-5 max-w-3xl overflow-x-auto">
   279→          <TabsTrigger value="overview" className="text-xs">
   280→            Genel Durum
   281→          </TabsTrigger>
   282→          <TabsTrigger value="users" className="text-xs">
   283→            Kullanıcılar
   284→          </TabsTrigger>
   285→          <TabsTrigger value="tasks" className="text-xs">
   286→            Görev & Güvence
   287→          </TabsTrigger>
   288→          <TabsTrigger value="console" className="text-xs">
   289→            Sistem Konsolu
   290→          </TabsTrigger>
   291→          <TabsTrigger value="admin-info" className="text-xs">
   292→            Yönetici
   293→          </TabsTrigger>
   294→        </TabsList>
   295→
   296→        {/* Overview Tab */}
   297→        <TabsContent value="overview" className="mt-5 space-y-5">
   298→          {stats && (
   299→            <>
   300→              <div className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   301→                {[
   302→                  { label: 'Kullanıcı', value: stats.totalUsers, icon: Users, color: 'text-blue-600' },
   303→                  { label: 'Görev', value: stats.totalTasks, icon: ClipboardList, color: 'text-purple-600' },
   304→                  { label: 'Bekleyen', value: stats.pendingTasks, icon: Info, color: 'text-amber-600' },
   305→                  { label: 'Tamamlanan', value: stats.completedTasks, icon: UserCheck, color: 'text-green-600' },
   306→                  { label: 'İşlem Hacmi', value: formatTL(stats.totalVolume), icon: DollarSign, color: 'text-red-600' },
   307→                ].map((item) => (
   308→                  <Card key={item.label}>
   309→                    <CardContent className="p-4 text-center">
   310→                      <item.icon className={`w-5 h-5 ${item.color} mx-auto mb-1.5`} />
   311→                      <p className="text-xs text-muted-foreground">{item.label}</p>
   312→                      <p className="text-lg font-bold mt-0.5">{item.value}</p>
   313→                    </CardContent>
   314→                  </Card>
   315→                ))}
   316→              </div>
   317→
   318→              {/* Financial summary */}
   319→              <Card>
   320→                <CardHeader className="pb-2">
   321→                  <CardTitle className="text-sm">Finansal Özet</CardTitle>
   322→                </CardHeader>
   323→                <CardContent>
   324→                  <div className="grid grid-cols-2 gap-4">
   325→                    <div className="p-3 rounded-lg bg-green-50 dark:bg-green-950/30">
   326→                      <p className="text-xs text-muted-foreground">Toplam İşlem Hacmi</p>
   327→                      <p className="text-lg font-bold text-green-600 dark:text-green-400">
   328→                        {formatTL(stats.totalVolume)}
   329→                      </p>
   330→                    </div>
   331→                    <div className="p-3 rounded-lg bg-amber-50 dark:bg-amber-950/30">
   332→                      <p className="text-xs text-muted-foreground">Escrow Hacmi</p>
   333→                      <p className="text-lg font-bold text-amber-600 dark:text-amber-400">
   334→                        {formatTL(stats.escrowVolume)}
   335→                      </p>
   336→                    </div>
   337→                  </div>
   338→                </CardContent>
   339→              </Card>
   340→            </>
   341→          )}
   342→        </TabsContent>
   343→
   344→        {/* Users Tab */}
   345→        <TabsContent value="users" className="mt-5">
   346→          <Card>
   347→            <CardContent className="p-0">
   348→              <div className="max-h-[500px] overflow-y-auto">
   349→                <Table>
   350→                  <TableHeader>
   351→                    <TableRow>
   352→                      <TableHead className="text-xs">Kullanıcı</TableHead>
   353→                      <TableHead className="text-xs">Rol</TableHead>
   354→                      <TableHead className="text-xs">T.C.</TableHead>
   355→                      <TableHead className="text-xs">Bakiye</TableHead>
   356→                      <TableHead className="text-xs">Şehir</TableHead>
   357→                      <TableHead className="text-xs text-right">İşlemler</TableHead>
   358→                    </TableRow>
   359→                  </TableHeader>
   360→                  <TableBody>
   361→                    {users.map((user) => (
   362→                      <TableRow key={user.id}>
   363→                        <TableCell>
   364→                          <div>
   365→                            <p className="text-sm font-medium">{user.fullName}</p>
   366→                            <p className="text-xs text-muted-foreground">{user.email}</p>
   367→                          </div>
   368→                        </TableCell>
   369→                        <TableCell>
   370→                          <Badge variant={user.role === 'admin' ? 'destructive' : 'secondary'} className="text-[10px]">
   371→                            {user.role === 'admin' ? 'Admin' : 'Üye'}
   372→                          </Badge>
   373→                        </TableCell>
   374→                        <TableCell>
   375→                          <Switch
   376→                            checked={user.tcVerified}
   377→                            onCheckedChange={() => handleToggleTC(user)}
   378→                            className="data-[state=checked]:bg-green-500"
   379→                          />
   380→                        </TableCell>
   381→                        <TableCell className="text-sm font-medium">
   382→                          {formatTL(user.walletBalance)}
   383→                        </TableCell>
   384→                        <TableCell className="text-sm text-muted-foreground">{user.city}</TableCell>
   385→                        <TableCell>
   386→                          <div className="flex items-center justify-end gap-1">
   387→                            <BalanceAdjustButton userId={user.id} onAdjust={handleAdjustBalance} />
   388→                            {user.role !== 'admin' && (
   389→                              <AlertDialog>
   390→                                <AlertDialogTrigger asChild>
   391→                                  <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500">
   392→                                    <Trash2 className="w-3.5 h-3.5" />
   393→                                  </Button>
   394→                                </AlertDialogTrigger>
   395→                                <AlertDialogContent>
   396→                                  <AlertDialogHeader>
   397→                                    <AlertDialogTitle>Kullanıcı Sil</AlertDialogTitle>
   398→                                    <AlertDialogDescription>
   399→                                      <strong>{user.fullName}</strong> kullanıcısını ve tüm
   400→                                      ilişkili verileri silmek istediğinize emin misiniz?
   401→                                    </AlertDialogDescription>
   402→                                  </AlertDialogHeader>
   403→                                  <AlertDialogFooter>
   404→                                    <AlertDialogCancel>İptal</AlertDialogCancel>
   405→                                    <AlertDialogAction onClick={() => handleDeleteUser(user.id, user.fullName)}>
   406→                                      Sil
   407→                                    </AlertDialogAction>
   408→                                  </AlertDialogFooter>
   409→                                </AlertDialogContent>
   410→                              </AlertDialog>
   411→                            )}
   412→                          </div>
   413→                        </TableCell>
   414→                      </TableRow>
   415→                    ))}
   416→                  </TableBody>
   417→                </Table>
   418→              </div>
   419→            </CardContent>
   420→          </Card>
   421→        </TabsContent>
   422→
   423→        {/* Tasks Tab */}
   424→        <TabsContent value="tasks" className="mt-5">
   425→          <Card>
   426→            <CardContent className="p-0">
   427→              <div className="max-h-[500px] overflow-y-auto">
   428→                <Table>
   429→                  <TableHeader>
   430→                    <TableRow>
   431→                      <TableHead className="text-xs">Görev</TableHead>
   432→                      <TableHead className="text-xs">Kategori</TableHead>
   433→                      <TableHead className="text-xs">Durum</TableHead>
   434→                      <TableHead className="text-xs">Ücret</TableHead>
   435→                      <TableHead className="text-xs">Atanan</TableHead>
   436→                      <TableHead className="text-xs text-right">İşlemler</TableHead>
   437→                    </TableRow>
   438→                  </TableHeader>
   439→                  <TableBody>
   440→                    {allTasks.map((task) => {
   441→                      const status = statusMap[task.status] || statusMap.pending;
   442→                      return (
   443→                        <TableRow key={task.id}>
   444→                          <TableCell>
   445→                            <p className="text-sm font-medium max-w-48 truncate">{task.title}</p>
   446→                            <p className="text-xs text-muted-foreground">{task.city}</p>
   447→                          </TableCell>
   448→                          <TableCell className="text-xs">{task.category}</TableCell>
   449→                          <TableCell>
   450→                            <Select
   451→                              value={task.status}
   452→                              onValueChange={(v) => handleTaskStatus(task.id, v)}
   453→                            >
   454→                              <SelectTrigger className="h-7 text-xs w-32">
   455→                                <SelectValue />
   456→                              </SelectTrigger>
   457→                              <SelectContent>
   458→                                <SelectItem value="pending">Bekliyor</SelectItem>
   459→                                <SelectItem value="assigned">Atandı</SelectItem>
   460→                                <SelectItem value="in_progress">Devam Ediyor</SelectItem>
   461→                                <SelectItem value="completed">Tamamlandı</SelectItem>
   462→                                <SelectItem value="cancelled">İptal</SelectItem>
   463→                              </SelectContent>
   464→                            </Select>
   465→                          </TableCell>
   466→                          <TableCell className="text-sm font-semibold">
   467→                            {formatTL(task.fee)}
   468→                          </TableCell>
   469→                          <TableCell className="text-xs text-muted-foreground">
   470→                            {task.assignedToName || '-'}
   471→                          </TableCell>
   472→                          <TableCell className="text-right">
   473→                            <AssignTaskerSelect
   474→                              task={task}
   475→                              users={users.filter((u) => u.role !== 'admin')}
   476→                              onAssign={(userId, userName) =>
   477→                                handleTaskStatus(task.id, 'assigned', userId, userName)
   478→                              }
   479→                            />
   480→                          </TableCell>
   481→                        </TableRow>
   482→                      );
   483→                    })}
   484→                  </TableBody>
   485→                </Table>
   486→              </div>
   487→            </CardContent>
   488→          </Card>
   489→        </TabsContent>
   490→
   491→        {/* Console Tab */}
   492→        <TabsContent value="console" className="mt-5">
   493→          <Card className="border-green-200 dark:border-green-800">
   494→            <CardHeader className="pb-2">
   495→              <CardTitle className="text-sm flex items-center gap-2">
   496→                <Terminal className="w-4 h-4 text-green-600" />
   497→                Sistem Konsolu & Log
   498→              </CardTitle>
   499→            </CardHeader>
   500→            <CardContent>
   501→              <div className="bg-black/90 dark:bg-black rounded-lg p-4 font-mono text-[11px] text-green-400 max-h-80 overflow-y-auto">
   502→                {systemLogs.length === 0 ? (
   503→                  <p className="text-green-600/50">Sistem logları bekleniyor...</p>
   504→                ) : (
   505→                  systemLogs.map((log, i) => (
   506→                    <div key={i} className="py-0.5 border-b border-green-900/30 last:border-0">
   507→                      {log}
   508→                    </div>
   509→                  ))
   510→                )}
   511→                <div ref={logEndRef} />
   512→              </div>
   513→            </CardContent>
   514→          </Card>
   515→        </TabsContent>
   516→
   517→        {/* Admin Info Tab */}
   518→        <TabsContent value="admin-info" className="mt-5">
   519→          {currentUser && (
   520→            <Card className="max-w-lg">
   521→              <CardContent className="p-6 space-y-4">
   522→                <div className="flex items-center gap-4">
   523→                  <div className="w-16 h-16 rounded-2xl bg-gradient-to-br from-blue-600 to-indigo-700 flex items-center justify-center text-white text-xl font-bold">
   524→                    {currentUser.fullName.split(' ').map((n) => n[0]).join('').slice(0, 2)}
   525→                  </div>
   526→                  <div>
   527→                    <h3 className="text-lg font-bold">{currentUser.fullName}</h3>
   528→                    <Badge variant="destructive" className="text-[10px] mt-1">
   529→                      Sistem Yöneticisi
   530→                    </Badge>
   531→                  </div>
   532→                </div>
   533→                <Separator />
