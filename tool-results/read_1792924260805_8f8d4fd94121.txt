    52→export default function TurkeyMap({ tasks, onCitySelect, onTaskClick, selectedCity: selectedCityProp }: TurkeyMapProps) {

    55→  const [isZoomed, setIsZoomed] = useState(false);

    58→  const markersRef = useRef<any>(null);
    59→  const boundaryRef = useRef<any>(null);
    60→  const taskMarkersRef = useRef<any[]>([]);
    61→  const cityPolygonsRef = useRef<any[]>([]);
    62→  const leafletRef = useRef<any>(null);
    63→  const tileLayerRef = useRef<any>(null);
    64→  const refLayerRef = useRef<any>(null);
    65→  const { theme } = useTheme();
    66→
    67→  const { currentUser, addLog } = useAppStore();
    68→  const isAdmin = currentUser?.role === 'admin';
    69→  const defaultBoundsRef = useRef<any>([[35.5, 25.5], [42.5, 45.0]]);
    70→
    71→  const isZoomedRef = useRef(isZoomed);
    72→  useEffect(() => {

    75→
    76→  const isAdminRef = useRef(isAdmin);
    77→  useEffect(() => {
    78→    isAdminRef.current = isAdmin;
    79→  }, [isAdmin]);
    80→
    81→
    82→  useEffect(() => {
    83→    if (!tileLayerRef.current) return;
    84→    const isDark = theme === 'dark';
    85→    tileLayerRef.current.setUrl(
    86→      isDark
    87→        ? 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Dark_Gray_Base/MapServer/tile/{z}/{y}/{x}'
    88→        : 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Light_Gray_Base/MapServer/tile/{z}/{y}/{x}'
    89→    );
    90→    if (refLayerRef.current) {
    91→      refLayerRef.current.setUrl(
    92→        isDark
    93→          ? 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Dark_Gray_Reference/MapServer/tile/{z}/{y}/{x}'
    94→          : 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Light_Gray_Reference/MapServer/tile/{z}/{y}/{x}'
    95→      );
    96→    }
    97→  }, [theme]);
    98→
    99→  const activeOnlyTasks = useMemo(() => {
   100→    if (!tasks || !Array.isArray(tasks)) return [];
   101→    return tasks.filter((t) => t.status !== 'completed' && t.status !== 'cancelled');
   102→  }, [tasks]);
   103→
   104→  const tasksByCity = useMemo(() => {
   105→    const map: Record<string, Task[]> = {};
   106→    activeOnlyTasks.forEach((t) => {
   107→      if (!map[t.city]) map[t.city] = [];
   108→      map[t.city].push(t);
   109→    });
   110→    return map;
   111→  }, [activeOnlyTasks]);
   112→
   113→  const activeCities = useMemo(() => Object.keys(tasksByCity), [tasksByCity]);
   114→
   115→  const activeCitiesRef = useRef(activeCities);
   116→  useEffect(() => {
   117→    activeCitiesRef.current = activeCities;
   118→  }, [activeCities]);
   119→
   120→  const tasksByCityRef = useRef(tasksByCity);
   121→  useEffect(() => {
   122→    tasksByCityRef.current = tasksByCity;
   123→  }, [tasksByCity]);
   124→
   125→  const selectedCityTasks = useMemo(() => {
   126→    if (!selectedCity) return [];
   127→    return tasksByCity[selectedCity] || [];
   128→  }, [selectedCity, tasksByCity]);
   129→
   130→  useEffect(() => {
   131→    if (!mapRef.current || ready) return;
   132→    let cancelled = false;
   133→
   134→    (async () => {
   135→      const leafletMod = await import('leaflet');
   136→      const L = leafletMod.default;
   137→      leafletRef.current = L;
   138→      if (cancelled || !mapRef.current) return;
   139→
   140→      const map = L.map(mapRef.current, {
   141→        zoomControl: false,
   142→        attributionControl: false,
   143→        minZoom: 4,
   144→        maxZoom: 14,
   145→      });
   146→
   147→      // Fit bounds based on user session or admin global default setting
   148→      const savedUserView = typeof window !== 'undefined' ? sessionStorage.getItem('benimicinyap_user_map_view') : null;
   149→      let initialViewLoaded = false;
   150→
   151→      if (savedUserView) {
   152→        try {
   153→          const { center, zoom } = JSON.parse(savedUserView);
   154→          map.setView(center, zoom);
   155→          initialViewLoaded = true;
   156→        } catch (e) {
   157→          console.warn("Error parsing saved view:", e);
   158→        }
   159→      }
   160→
   161→      if (!initialViewLoaded) {
   162→        fetch('/api/admin/settings')
   163→          .then((res) => res.json())
   164→          .then((data) => {
   165→            if (data.success && Array.isArray(data.settings)) {
   166→              const boundsSetting = data.settings.find((s: any) => s.key === 'map_default_bounds');
   167→              if (boundsSetting && boundsSetting.value) {
   168→                try {
   169→                  const bounds = JSON.parse(boundsSetting.value);
   170→                  defaultBoundsRef.current = bounds;
   171→                  map.fitBounds(bounds, { padding: [10, 10] });
   172→                  return;
   173→                } catch {}
   174→              }
   175→            }
   176→            map.fitBounds([[35.5, 25.5], [42.5, 45.0]], { padding: [10, 10] });
   177→          })
   178→          .catch(() => {
   179→            map.fitBounds([[35.5, 25.5], [42.5, 45.0]], { padding: [10, 10] });
   180→          });
   181→      }
   182→
   183→      // Minimalist Canvas tile layer (Clean, sleek, zero green/river clutter, 100% free)
   184→      const isDarkInit = typeof document !== 'undefined' && document.documentElement.classList.contains('dark');
   185→      const tileLayer = L.tileLayer(
   186→        isDarkInit
   187→          ? 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Dark_Gray_Base/MapServer/tile/{z}/{y}/{x}'
   188→          : 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Light_Gray_Base/MapServer/tile/{z}/{y}/{x}',
   189→        {
   190→          maxZoom: 16,
   191→          attribution: '&copy; Esri, DeLorme, NAVTEQ',
   192→        }
   193→      ).addTo(map);
   194→      tileLayerRef.current = tileLayer;
   195→
   196→      // Reference layer for city & province labels
   197→      const refLayer = L.tileLayer(
   198→        isDarkInit
   199→          ? 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Dark_Gray_Reference/MapServer/tile/{z}/{y}/{x}'
   200→          : 'https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Light_Gray_Reference/MapServer/tile/{z}/{y}/{x}',
   201→        {
   202→          maxZoom: 16,
   203→          pane: 'overlayPane',
   204→          zIndex: 350,
   205→        }
   206→      ).addTo(map);
   207→      refLayerRef.current = refLayer;
   208→
   209→      L.control.zoom({ position: 'bottomright' }).addTo(map);
   210→
   211→      // Listen for user pan/zoom to save their view state
   212→      map.on('moveend zoomend', () => {
   213→        if (!isZoomedRef.current) {
   214→          try {
   215→            const bounds = map.getBounds();
   216→            const boundsVal = JSON.stringify([
   217→              [bounds.getSouthWest().lat, bounds.getSouthWest().lng],
   218→              [bounds.getNorthEast().lat, bounds.getNorthEast().lng]
   219→            ]);
   220→
   221→            const viewState = {
   222→              center: [map.getCenter().lat, map.getCenter().lng],
   223→              zoom: map.getZoom(),
   224→            };
   225→            sessionStorage.setItem('benimicinyap_user_map_view', JSON.stringify(viewState));
   226→
   227→            // Autosave admin default bounds on change (eliminates manual 'sabitle' button)
   228→            if (isAdminRef.current) {
   229→              fetch('/api/admin/settings', {
   230→                method: 'PUT',
   231→                headers: { 'Content-Type': 'application/json' },
   232→                body: JSON.stringify({
   233→                  settings: [
   234→                    { key: 'map_default_bounds', value: boundsVal }
   235→                  ]
   236→                })
   237→              }).catch(() => {});
   238→            }
   239→          } catch {}
   240→        }
   241→      });
   242→
   243→      // 81 İl Sınırları Katmanı (Tüm Türkiye İlleri - Tıklanabilir & Vurgulanabilir)
   244→      L.geoJSON(trGeoJSON as any, {
   245→        style: (feature: any) => {
   246→          const cityName = feature?.properties?.name;
   247→          const hasTasks = cityName && activeCitiesRef.current.includes(cityName);
   248→          return {
   249→            color: hasTasks ? '#2563eb' : '#64748b',
   250→            weight: hasTasks ? 1.5 : 0.9,
   251→            fillColor: hasTasks ? '#3b82f6' : '#94a3b8',
   252→            fillOpacity: hasTasks ? 0.15 : 0.05,
   253→            opacity: 0.75,
   254→          };
   255→        },
   256→        onEachFeature: (feature: any, layer: any) => {
   257→          const cityName = feature?.properties?.name;
   258→          if (!cityName) return;
   259→          const cityGeo = turkeyCitiesGeo.find((c) => normalizeName(c.name) === normalizeName(cityName));
   260→
   261→          layer.on({
   262→            mouseover: (e: any) => {
   263→              const target = e.target;
   264→              target.setStyle({
   265→                weight: 2.5,
   266→                color: '#1d4ed8',
   267→                fillOpacity: 0.28,
   268→              });
   269→              if (!L.Browser.ie && !L.Browser.opera && !L.Browser.edge) {
   270→                target.bringToFront();
   271→              }
   272→            },
   273→            mouseout: (e: any) => {
   274→              const target = e.target;
   275→              const hasTasks = activeCitiesRef.current.includes(cityName);
   276→              target.setStyle({
   277→                color: hasTasks ? '#2563eb' : '#64748b',
   278→                weight: hasTasks ? 1.5 : 0.9,
   279→                fillColor: hasTasks ? '#3b82f6' : '#94a3b8',
   280→                fillOpacity: hasTasks ? 0.15 : 0.05,
   281→                opacity: 0.75,
   282→              });
   283→            },
   284→            click: () => {
   285→              if (cityGeo) {
   286→                flyToCity(L, map, cityGeo, tasksByCityRef.current, onCitySelect, onTaskClick, taskMarkersRef, boundaryRef);
   287→                setSelectedCity(cityGeo.name);
   288→                setIsZoomed(true);
   289→              }
   290→            },
   291→          });
   292→          cityPolygonsRef.current.push(layer);
   293→        },
   294→      }).addTo(map);
   295→
   296→      // Add markers
   297→      const markers = L.layerGroup().addTo(map);
   298→      turkeyCitiesGeo.forEach((city) => {
   299→        const hasTasks = activeCities.includes(city.name);
   300→        const cityTasks = hasTasks ? (tasksByCity[city.name] || []) : [];
   301→        const taskCount = cityTasks.length;
   302→        const statusBreakdown = getCityStatusBreakdown(cityTasks);
   303→
   304→        const marker = L.marker([city.lat, city.lng], {
   305→          icon: hasTasks
   306→            ? createStatusIcon(L, taskCount, statusBreakdown)
   307→            : createInactiveIcon(L),
   308→          zIndexOffset: hasTasks ? 1000 : 1,
   309→        });
   310→
   311→        if (hasTasks) {
   312→          marker.bindTooltip(
   313→            buildCityTooltip(city.name, taskCount, statusBreakdown),
   314→            { direction: 'top', offset: [0, -18], className: 'city-tooltip-clean' }
   315→          );
   316→        }
   317→
   318→        marker.on('click', () => {
   319→          if (!mapObjRef.current) return;
   320→          flyToCity(L, map, city, tasksByCity, onCitySelect, onTaskClick, taskMarkersRef, boundaryRef);
   321→          setSelectedCity(city.name);
   322→          setIsZoomed(true);
   323→        });
   324→        markers.addLayer(marker);
   325→      });
   326→      markersRef.current = markers;
   327→      mapObjRef.current = map;
   328→      setReady(true);
   329→    })();
   330→
   331→    return () => {
   332→      cancelled = true;
   333→      if (mapObjRef.current) {
   334→        mapObjRef.current.remove();
   335→        mapObjRef.current = null;
   336→      }
   337→      setReady(false);
   338→    };
   339→  }, []);
   340→
   341→  // Update markers when tasks change
   342→  useEffect(() => {
   343→    if (!ready || !markersRef.current || !mapObjRef.current || !leafletRef.current) return;
   344→    const L = leafletRef.current;
   345→    markersRef.current.clearLayers();
   346→
   347→    turkeyCitiesGeo.forEach((city) => {
   348→      const hasTasks = activeCities.includes(city.name);
   349→      const cityTasks = hasTasks ? (tasksByCity[city.name] || []) : [];
   350→      const taskCount = cityTasks.length;
   351→      const statusBreakdown = getCityStatusBreakdown(cityTasks);
   352→
   353→      const marker = L.marker([city.lat, city.lng], {
   354→        icon: hasTasks
   355→          ? createStatusIcon(L, taskCount, statusBreakdown)
   356→          : createInactiveIcon(L),
   357→        zIndexOffset: hasTasks ? 1000 : 1,
   358→      });
   359→
   360→      if (hasTasks) {
   361→        marker.bindTooltip(
   362→          buildCityTooltip(city.name, taskCount, statusBreakdown),
   363→          { direction: 'top', offset: [0, -18], className: 'city-tooltip-clean' }
   364→        );
   365→      }
   366→
   367→      marker.on('click', () => {
   368→        if (!mapObjRef.current) return;
   369→        flyToCity(L, mapObjRef.current, city, tasksByCity, onCitySelect, onTaskClick, taskMarkersRef, boundaryRef);
   370→        setSelectedCity(city.name);
   371→        setIsZoomed(true);
   372→      });
   373→      markersRef.current.addLayer(marker);
   374→    });
   375→  }, [ready, activeCities, tasksByCity, onCitySelect, onTaskClick]);
   376→
   377→  const handleBackToTurkey = useCallback(() => {
   378→    if (!mapObjRef.current || !leafletRef.current) return;
   379→    setSelectedCity(null);
   380→    setIsZoomed(false);
   381→    onCitySelect?.('');
   382→    if (boundaryRef.current) {
   383→      mapObjRef.current.removeLayer(boundaryRef.current);
   384→      boundaryRef.current = null;
   385→    }
   386→    taskMarkersRef.current.forEach((m) => {
   387→      if (mapObjRef.current!.hasLayer(m)) mapObjRef.current!.removeLayer(m);
   388→    });
   389→    taskMarkersRef.current = [];
   390→    cityPolygonsRef.current.forEach((p: any) => {
   391→      const cityName = p?.feature?.properties?.name;
   392→      const hasTasks = cityName && activeCitiesRef.current.includes(cityName);
   393→      p.setStyle({
   394→        color: hasTasks ? '#2563eb' : '#64748b',
   395→        weight: hasTasks ? 1.5 : 0.9,
   396→        fillColor: hasTasks ? '#3b82f6' : '#94a3b8',
   397→        fillOpacity: hasTasks ? 0.15 : 0.05,
   398→        opacity: 0.75,
   399→      });
   400→    });
   401→    mapObjRef.current.flyToBounds(defaultBoundsRef.current, { duration: 1.2, padding: [10, 10], easeLinearity: 0.25 });
   402→  }, [onCitySelect]);
   403→
   404→  // Sync with external selectedCity prop
   405→  useEffect(() => {
   406→    if (!ready || selectedCityProp === undefined) return;
   407→    const L = leafletRef.current;
   408→    if (!L) return;
   409→    if (selectedCityProp && selectedCityProp !== selectedCity) {
   410→      const city = turkeyCitiesGeo.find((c) => c.name === selectedCityProp);
   411→      if (city && mapObjRef.current) {
   412→        flyToCity(L, mapObjRef.current, city, tasksByCity, onCitySelect, onTaskClick, taskMarkersRef, boundaryRef);
   413→        setSelectedCity(selectedCityProp);
   414→        setIsZoomed(true);
   415→      }
   416→    } else if (!selectedCityProp && selectedCity && mapObjRef.current) {
   417→      handleBackToTurkey();
   418→    }
   419→  }, [selectedCityProp, ready, handleBackToTurkey, selectedCity, tasksByCity, onCitySelect, onTaskClick]);
   420→
   421→  const formatTL = (amount: number) =>
   422→    new Intl.NumberFormat('tr-TR').format(amount) + ' ₺';
   423→
   424→  // Active (non-completed, non-cancelled) tasks for sidebar
   425→  const activeTasks = useMemo(() => {
   426→    if (!tasks || !Array.isArray(tasks)) return [];
   427→    return tasks.filter((t) => t.status === 'pending' || t.status === 'assigned' || t.status === 'in_progress');
   428→  }, [tasks]);
   429→
   430→  // Group active tasks by city
   431→  const activeTasksByCity = useMemo(() => {
   432→    const map: Record<string, Task[]> = {};
   433→    activeTasks.forEach((t) => {
   434→      if (!map[t.city]) map[t.city] = [];
   435→      map[t.city].push(t);
   436→    });
   437→    return map;
   438→  }, [activeTasks]);
   439→
   440→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   441→
   442→  return (
   443→    <div className="relative w-full flex flex-col">
   444→      {/* Map + Sidebar Row */}
   445→      <div className="flex flex-col md:flex-row gap-3">
   446→      {/* Left Sidebar - Active Task List (desktop only) */}
   447→      <div className={`${sidebarCollapsed ? 'md:w-0 md:opacity-0 md:overflow-hidden' : 'md:w-80 shrink-0'} hidden md:flex transition-all duration-300`}>
   448→        <div className="glass-card rounded-xl shadow-sm h-[420px] flex flex-col overflow-hidden w-full">
   449→          {/* Sidebar Header */}
   450→          <div className="p-3 border-b border-slate-100 dark:border-slate-800 bg-slate-50/50 dark:bg-white/5">
   451→            <div className="flex items-center justify-between">
   452→              <div className="flex items-center gap-2">
   453→                <div className="w-6 h-6 rounded-lg bg-green-100 dark:bg-green-900/40 flex items-center justify-center">
   454→                  <span className="text-[10px] font-bold text-green-700 dark:text-green-300">{activeTasks.length}</span>
   455→                </div>
   456→                <h4 className="font-semibold text-sm text-slate-800 dark:text-slate-100">Aktif Görevler</h4>
   457→              </div>
   458→              <button
   459→                onClick={() => setSidebarCollapsed(true)}
   460→                className="w-6 h-6 rounded-md hover:bg-slate-200 dark:hover:bg-slate-700 flex items-center justify-center transition-colors"
   461→              >
   462→                <span className="text-xs text-muted-foreground">◂</span>
   463→              </button>
   464→            </div>
   465→          </div>
   466→
   467→          {/* Task List */}
   468→          <div className="flex-1 overflow-y-auto">
   469→            {Object.entries(activeTasksByCity).map(([city, cityTasks]) => (
   470→              <div key={city}>
   471→                {/* City group header */}
   472→                <div className="px-3 py-2 bg-slate-50/80 dark:bg-slate-800/50 sticky top-0 z-10">
   473→                  <div className="flex items-center gap-1.5">
   474→                    <MapPin className="w-3 h-3 text-blue-500" />
   475→                    <span className="text-[11px] font-bold text-slate-600 dark:text-slate-300 uppercase tracking-wider">{city}</span>
   476→                    <Badge variant="secondary" className="text-[9px] ml-auto px-1.5 py-0">{cityTasks.length}</Badge>
   477→                  </div>
   478→                </div>
   479→
   480→                {/* Tasks under this city */}
   481→                {cityTasks.map((task) => (

   742→function getCityStatusBreakdown(tasks: Task[]): StatusBreakdown {
   743→  const counts = { pending: 0, assigned: 0, in_progress: 0, completed: 0, cancelled: 0 };
   744→  tasks.forEach((t) => {
   745→    const s = t.status as keyof typeof counts;
   746→    if (s in counts) counts[s]++;
   747→  });
   748→  // Dominant color: prefer active statuses first
   749→  const priority: Array<keyof typeof counts> = ['in_progress', 'pending', 'assigned', 'completed', 'cancelled'];
   750→  const dominant = priority.find((s) => counts[s] > 0) || 'pending';
   751→  return { ...counts, dominantColor: STATUS_COLORS[dominant]?.bg || '#3b82f6' };
   752→}

   754→function createStatusIcon(L: any, count: number, breakdown: StatusBreakdown) {
   755→  const displayCount = count > 99 ? '99+' : String(count);
   756→  const size = 34; // compact task count badge
   757→  const r = size / 2 - 3;
   758→  const cx = size / 2;
   759→  const cy = size / 2;
   760→
   761→  // Build pie segments for the ring based on status breakdown
   762→  const activeStatuses: Array<{ key: string; count: number }> = [
   763→    { key: 'pending', count: breakdown.pending },
   764→    { key: 'in_progress', count: breakdown.in_progress },
   765→    { key: 'assigned', count: breakdown.assigned },
   766→  ].filter((s) => s.count > 0);
   767→
   768→  const total = activeStatuses.reduce((s, x) => s + x.count, 0) || 1;
   769→  let segmentsSvg = '';
   770→  let startAngle = -90; // Start from top
   771→
   772→  if (activeStatuses.length === 1) {
   773→    // Single color circle
   774→    const color = STATUS_COLORS[activeStatuses[0].key]?.bg || breakdown.dominantColor;
   775→    segmentsSvg = `<circle cx="${cx}" cy="${cy}" r="${r}" fill="${color}" stroke="white" stroke-width="1.5" class="dark:stroke-[#0B0C16]"/>`;
   776→  } else {
   777→    // Multi-color ring segments
   778→    segmentsSvg = `<circle cx="${cx}" cy="${cy}" r="${r}" fill="${breakdown.dominantColor}" stroke="white" stroke-width="1.5" class="dark:stroke-[#0B0C16]"/>`;
   779→    activeStatuses.forEach((seg) => {
   780→      const angle = (seg.count / total) * 360;
   781→      const endAngle = startAngle + angle;
   782→      const rad = Math.PI / 180;
   783→      const x1 = cx + r * Math.cos(startAngle * rad);
   784→      const y1 = cy + r * Math.sin(startAngle * rad);
   785→      const x2 = cx + r * Math.cos(endAngle * rad);
   786→      const y2 = cy + r * Math.sin(endAngle * rad);
   787→      const largeArc = angle > 180 ? 1 : 0;
   788→      const color = STATUS_COLORS[seg.key]?.bg || '#888';
   789→      segmentsSvg += `<path d="M${cx},${cy} L${x1},${y1} A${r},${r} 0 ${largeArc},1 ${x2},${y2} Z" fill="${color}" stroke="white" stroke-width="0.5" class="dark:stroke-[#0B0C16]"/>`;
   790→      startAngle = endAngle;
   791→    });
   792→  }
   793→
   794→  const isDark = typeof document !== 'undefined' && document.documentElement.classList.contains('dark');
   795→  const glowFilter = isDark ? `
   796→    <defs>
   797→      <filter id="neon-glow" x="-30%" y="-30%" width="160%" height="160%">
   798→        <feGaussianBlur stdDeviation="2.5" result="blur" />
   799→        <feComponentTransfer in="blur" result="boost">
   800→          <feFuncA type="linear" slope="2"/>
   801→        </feComponentTransfer>
   802→        <feMerge>
   803→          <feMergeNode in="boost" />
   804→          <feMergeNode in="SourceGraphic" />
   805→        </feMerge>
   806→      </filter>
   807→    </defs>
   808→  ` : '';
   809→
   810→  const filterAttr = isDark ? 'filter="url(#neon-glow)"' : '';
   811→
   812→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${size}" height="${size}" viewBox="0 0 ${size} ${size}">
   813→    ${glowFilter}
   814→    <g ${filterAttr}>
   815→      ${segmentsSvg}
   816→    </g>
   817→    <circle cx="${cx}" cy="${cy}" r="${r - 5}" fill="white" class="dark:fill-[#0B0C16]"/>
   818→    <text x="${cx}" y="${cy + 0.5}" text-anchor="middle" dominant-baseline="middle" fill="${isDark ? '#4ade80' : breakdown.dominantColor}" font-size="10" font-weight="900" font-family="system-ui" class="dark:fill-[#4ade80]">${displayCount}</text>
   819→  </svg>`;
   820→
   821→  return L.divIcon({
   822→    html: svg,
   823→    className: 'task-marker-label',
   824→    iconSize: [size, size],
   825→    iconAnchor: [size / 2, size / 2],
   826→  });
   827→}

   858→const normalizeName = (name: string) => {
   859→  return name
   860→    .toLowerCase()
   861→    .replace(/ğ/g, 'g')
   862→    .replace(/ü/g, 'u')

   867→    .replace(/â/g, 'a')
   868→    .replace(/î/g, 'i')

   870→    .replace(/[^a-z0-9]/g, '')
   871→    .trim();

   876→(trGeoJSON as any).features.forEach((f: any) => {

   881→function findGeoFeature(cityName: string): any | null {
   882→  const normalizedCity = normalizeName(cityName);
   883→
   884→  // 1. Exact match
   885→  if (geoJSONLookup[normalizedCity]) return geoJSONLookup[normalizedCity];
   886→
   887→  // 2. GeoJSON name is shorter prefix of city (e.g. "afyon" matches "afyonkarahisar")
   888→  for (const [key, feature] of Object.entries(geoJSONLookup)) {
   889→    if (normalizedCity.startsWith(key) || key.startsWith(normalizedCity)) {
   890→      return feature;
   891→    }
   892→  }
   893→
   894→  // 3. Substring match
   895→  for (const [key, feature] of Object.entries(geoJSONLookup)) {
   896→    if (normalizedCity.includes(key) || key.includes(normalizedCity)) {
   897→      return feature;
   898→    }
   899→  }
   900→
   901→  return null;
   902→}

   926→
   927→  if (taskMarkersRef?.current) {
   928→    taskMarkersRef.current.forEach((m) => {
   929→      if (map.hasLayer(m)) map.removeLayer(m);
   930→    });
   931→    taskMarkersRef.current = [];
   932→  }
   933→
   934→  // Find real detailed boundary feature in loaded GeoJSON using robust lookup
   935→  const feature = findGeoFeature(city.name);
   936→
   937→  let boundaryLayer: any;
   938→  if (feature) {
   939→    boundaryLayer = L.geoJSON(feature, {
   940→      style: {
   941→        color: '#ef4444',
   942→        fillColor: '#ef4444',
   943→        fillOpacity: 0.18,
   944→        weight: 4.5,

   947→      className: 'city-boundary-highlight',
   948→    }).addTo(map);
   949→  } else {
   950→    // Fallback to simplified boundary if not found
   951→    boundaryLayer = L.polygon(
   952→      city.bounds.map((b) => [b[0], b[1]]),
   953→      {
   954→        color: '#ef4444',
   955→        fillColor: '#ef4444',
   956→        fillOpacity: 0.18,
   957→        weight: 4.5,
   958→        dashArray: '10 5',
   959→        className: 'city-boundary-highlight',
   960→      }
   961→    ).addTo(map);
   962→  }
   963→
   964→  // fly to the bounds of the boundary layer dynamically for perfect responsive layout on mobile/desktop
   965→  try {
   966→    map.flyToBounds(boundaryLayer.getBounds(), { duration: 1.5, padding: [25, 25], easeLinearity: 0.25 });
   967→  } catch (e) {
   968→    console.warn("Could not fly to bounds:", e);
   969→    map.flyTo([city.lat, city.lng], CITY_ZOOM, { duration: 1.5, easeLinearity: 0.25 });
   970→  }
   971→
   972→  if (boundaryRef) {
   973→    boundaryRef.current = boundaryLayer;
   974→  }
   975→
   976→  // Task markers with fee labels, colored by status
   977→  const statusMarkerStyles: Record<string, { bg: string; shadow: string; label: string }> = {
   978→    pending:     { bg: '#f59e0b', shadow: 'rgba(245,158,11,0.45)', label: 'Bekleyen' },
   979→    assigned:    { bg: '#3b82f6', shadow: 'rgba(59,130,246,0.45)', label: 'Mevcut' },

   982→    cancelled:   { bg: '#ef4444', shadow: 'rgba(239,68,68,0.40)', label: 'İptal' },
   983→  };
   984→
   985→  const cityTasks = tasksByCity[city.name] || [];
   986→  cityTasks.forEach((task) => {
   987→    const offsetLat = city.lat + (Math.random() - 0.5) * 0.02;
   988→    const offsetLng = city.lng + (Math.random() - 0.5) * 0.02;
   989→    const style = statusMarkerStyles[task.status] || statusMarkerStyles.pending;
   990→
   991→    const taskMarker = L.marker([offsetLat, offsetLng], {

   993→        html: `<div style="background:${style.bg};color:white;padding:3px 10px;border-radius:14px;font-size:11px;font-weight:600;white-space:nowrap;box-shadow:0 2px 8px ${style.shadow};cursor:pointer;border:1.5px solid rgba(255,255,255,0.4);">${task.fee.toLocaleString('tr-TR')} ₺</div>`,
   994→        className: 'task-marker-label',
   995→        iconAnchor: [35, 14],
   996→      }),
   997→      zIndexOffset: 2000,
   998→    });
   999→
  1000→    taskMarker.bindTooltip(
  1001→      `<div style="max-width:220px;"><strong>${task.title}</strong><br/><span style="color:#666;">${task.category}</span><br/><span style="display:inline-block;margin-top:3px;background:${style.bg};color:white;padding:1px 7px;border-radius:8px;font-size:10px;font-weight:600;">${style.label}</span> &nbsp;<span style="color:${style.bg};font-weight:700;">${task.fee.toLocaleString('tr-TR')} ₺</span></div>`,
  1002→      { direction: 'top', offset: [0, -10], className: 'city-tooltip-clean' }
  1003→    );
  1004→
  1005→    taskMarker.on('click', () => onTaskClick?.(task));
  1006→    taskMarker.addTo(map);
  1007→    taskMarkersRef?.current?.push(taskMarker);
  1008→  });
